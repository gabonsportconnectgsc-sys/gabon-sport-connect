// ── GALERIE PHOTOS — max 10 par membre, propriétaire uniquement ──
const GALLERY_MAX = 10;
let _galleryCurrentUrl = null;
let _galleryCurrentIndex = -1;

function renderGallery(photos=[]){
  const container=document.getElementById('photo-gallery');
  if(!container) return;
  const label=document.getElementById('gallery-count-label');
  const profilePhoto=userProfile?.photoURL||null;
  if(label) label.textContent=photos.length+'/'+GALLERY_MAX+' photo'+(photos.length!==1?'s':'');
  const canAdd = photos.length < GALLERY_MAX;
  const addBtn = canAdd
    ? `<div class="gallery-item gallery-add" onclick="document.getElementById('gallery-photo-input').click()" title="Ajouter une photo">📷</div>`
    : '';
  const limitMsg = !canAdd
    ? `<div class="gallery-limit-msg">⚠️ Limite de ${GALLERY_MAX} photos atteinte. Supprimez-en une pour en ajouter.</div>`
    : '';
  const items=photos.map((url,i)=>{
    const isProfile = url === profilePhoto;
    const safeUrl=sanitizeUrl(url);
    const urlJs=esc(String(safeUrl).replace(/\\/g,'\\\\').replace(/'/g,"\\'"));
    return `<div class="gallery-item">
      ${isProfile?'<div class="gallery-is-profile-badge">✓ Profil</div>':''}
      <img src="${esc(safeUrl)}" alt="Photo ${i+1}" loading="lazy" onclick="openGalleryModal('${urlJs}',${i})">
      <div class="gallery-item-overlay">
        <div class="gallery-item-actions">
          <button class="gallery-action-btn" onclick="event.stopPropagation();gallerySetAsProfileDirect(${i})" title="Définir comme photo de profil">⭐</button>
          <button class="gallery-action-btn" onclick="event.stopPropagation();galleryDownloadDirect('${urlJs}')" title="Télécharger">⬇️</button>
          <button class="gallery-action-btn danger" onclick="event.stopPropagation();galleryDeleteDirect(${i})" title="Supprimer">🗑️</button>
        </div>
      </div>
    </div>`;
  }).join('');
  container.innerHTML=items+addBtn;
  const limitEl=document.getElementById('gallery-limit-info');
  if(limitEl) limitEl.innerHTML=limitMsg;
}

function openGalleryModal(url,idx){
  const safeUrl=sanitizeUrl(url);
  _galleryCurrentUrl=safeUrl;
  _galleryCurrentIndex=idx;
  document.getElementById('gallery-modal-img').src=safeUrl;
  document.getElementById('gallery-modal').classList.add('open');
}
window.openGalleryModal=openGalleryModal;

function closeGalleryModal(){
  document.getElementById('gallery-modal').classList.remove('open');
  document.getElementById('gallery-modal-img').src='';
  _galleryCurrentUrl=null;
  _galleryCurrentIndex=-1;
}
window.closeGalleryModal=closeGalleryModal;

async function gallerySetAsProfile(){
  if(_galleryCurrentUrl) await gallerySetAsProfileDirect(_galleryCurrentIndex);
}
window.gallerySetAsProfile=gallerySetAsProfile;

async function gallerySetAsProfileDirect(idx){
  const photos=userProfile?.photos||[];
  const url=photos[idx];
  if(!url||!db||!currentUser) return;
  try{
    await ensureFirebaseAuthViaSupabase();
    await updateDoc(doc(db,'users',currentUser.uid),{photoURL:url,updatedAt:serverTimestamp()});
    userProfile.photoURL=url;
    renderDashboard();renderProfile();
    toast('✅ Photo de profil mise à jour !','success');
    closeGalleryModal();
  }catch(e){ toast('Erreur : '+e.message,'error'); }
}
window.gallerySetAsProfileDirect=gallerySetAsProfileDirect;

function galleryDownload(){
  if(_galleryCurrentUrl) galleryDownloadDirect(_galleryCurrentUrl);
}
window.galleryDownload=galleryDownload;

function galleryDownloadDirect(url){
  const a=document.createElement('a');
  a.href=url;
  a.download='photo-gsc.jpg';
  a.target='_blank';
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
}
window.galleryDownloadDirect=galleryDownloadDirect;

async function galleryDelete(){
  if(_galleryCurrentIndex>=0) await galleryDeleteDirect(_galleryCurrentIndex);
}
window.galleryDelete=galleryDelete;

async function galleryDeleteDirect(idx){
  if(!confirm('Supprimer cette photo ?')) return;
  if(!db||!currentUser||!userProfile) return;
  const photos=[...(userProfile.photos||[])];
  photos.splice(idx,1);
  try{
    await ensureFirebaseAuthViaSupabase();
    await updateDoc(doc(db,'users',currentUser.uid),{photos,updatedAt:serverTimestamp()});
    userProfile.photos=photos;
    renderGallery(photos);
    toast('🗑️ Photo supprimée.','info');
    closeGalleryModal();
  }catch(e){ toast('Erreur : '+e.message,'error'); }
}
window.galleryDeleteDirect=galleryDeleteDirect;
