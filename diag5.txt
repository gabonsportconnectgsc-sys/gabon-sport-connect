function renderOwnDocuments(){
  const container=document.getElementById('documents-own-list');
  const card=document.getElementById('prof-documents-card');
  if(!container||!card||!userProfile) return;
  card.style.display=isVisitor?'none':'';
  const docs=Array.isArray(userProfile.documents)?userProfile.documents:[];
  const label=document.getElementById('documents-count-label');
  if(label) label.textContent=docs.length+' document'+(docs.length!==1?'s':'');
  if(!docs.length){ container.innerHTML='<p style="font-size:12px;color:var(--gray-txt);">Aucun document pour le moment.</p>'; return; }
  container.innerHTML=docs.map((d,i)=>`
    <div class="doc-detail-row">
      <a href="${esc(sanitizeUrl(d.url))}" target="_blank" rel="noopener" class="doc-detail-link">📎 ${esc(d.label||'Document')}</a>
      ${d.locked?'<span class="doc-detail-lock">🔒 Sous contrat</span>':''}
      <button class="doc-detail-btn" onclick="shareOwnDocument('${esc(userProfile.uid)}',${i})" ${d.locked?'disabled':''} title="Partager">🔗</button>
      <button class="doc-detail-btn doc-detail-danger" onclick="removeOwnDocument('${esc(userProfile.uid)}',${i})" ${d.locked?'disabled':''} title="Supprimer">🗑️</button>
    </div>`).join('');
}

function onOwnDocumentFilesSelected(input){
  const files=Array.from(input.files||[]);
  if(!files.length) return;
  const tooBig=files.filter(f=>f.size>10*1024*1024);
  if(tooBig.length){
    toast('❌ Fichier(s) trop volumineux (max 10 Mo chacun) : '+tooBig.map(f=>f.name).join(', '),'error');
    input.value='';
    return;
  }
  _pendingOwnDocFiles=files.map(f=>({file:f,title:f.name.replace(/\.[^/.]+$/,'')}));
  renderPendingOwnDocsPanel();
}
